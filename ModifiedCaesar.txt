import java.util.*;
class Prac1B {
String Encrypt(int shift, String line) {
String result = "";
int offset;
for (int i = 0; i < line.length(); i++) {
offset = (line.charAt(i) + shift) % 256;
result += (char)offset;
}
return result;
}
String Decrypt(int shift, String line) {
String result = "";
int offset;
for (int i = 0; i < line.length(); i++) {
offset = (line.charAt(i) - shift) % 256;
if (offset < 0) {
offset += 256;
}
result += (char)offset;
}
return result;
}
public static void main(String args[]) {
Scanner sc = new Scanner(System.in);
Prac1B b = new Prac1B();
System.out.println("Menu:\n1: Encryption\n2: Decryption");
int choice = sc.nextInt();
sc.nextLine();
System.out.println("Enter the Shift : ");
int shift = sc.nextInt();
sc.nextLine();
System.out.println("Enter the Line : ");
String line = sc.nextLine();
System.out.println("Result :");
if (choice == 1) {
System.out.println(b.Encrypt(shift, line));
}
else if (choice == 2) {
System.out.println(b.Decrypt(shift, line));
}
else {
System.out.println("Invalid choice");
}
}
}
